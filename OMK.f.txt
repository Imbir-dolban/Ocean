HEX
\ Регистры 328P
$B8 constant TWBR $B9 constant TWSR $BB constant TWDR $BC constant TWCR
$7A constant ADCSRA $78 constant ADCW $7C constant ADMUX

: i2c! ( -- ) $48 TWBR c! $01 TWSR c! $04 TWCR c! ;
: i2cw ( -- ) begin TWCR c@ $80 and until ;
: i2cs ( -- ) $A4 TWCR c! i2cw ;
: i2cp ( -- ) $94 TWCR c! ;
: i2cb! ( b -- ) TWDR c! $84 TWCR c! i2cw ;
: i2c_r ( -- b ) $C4 TWCR c! i2cw TWDR c@ ; \ ack
: i2c_rn ( -- b ) $84 TWCR c! i2cw TWDR c@ ; \ nack

$EE constant B_W $EF constant B_R 

\ Чтение 16 бит с барометра
: b@16 ( reg -- val ) 
    i2cs B_W i2cb! i2cb! i2cs B_R i2cb!
    i2c_r 8 lshift i2c_rn or i2cp ;

\ Тут типа ваздухан (работа с ветром)
: adci ( -- ) $40 ADMUX c! $87 ADCSRA c! ;
: w@ ( c -- n ) $0F and $40 or ADMUX c! $C7 ADCSRA c!
    begin ADCSRA c@ $40 and 0= until ADCW @ ;

VARIABLE P VARIABLE W
$05 constant IMB_CALIB 

\ Опрос и вывод
: smp ( -- ) $F6 b@16 P ! 0 w@ IMB_CALIB + W ! ;
: rep ( -- ) CR ." [OMK] P:" P @ U. ."  W:" W @ U. ;

\ Точка входа
: main ( -- ) i2c! adci CR ." OMK v0.1 LOADED."
    begin smp rep 1000 ms KEY? until ;

main
